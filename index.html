സിസ്റ്റം ഓഫ് ചെയ്താലും ഡാറ്റ നഷ്ടപ്പെടാതിരിക്കാൻ നിങ്ങളുടെ Google Apps Script Web App URL ഇവിടെ നൽകി സേവ് ചെയ്യുക.

Google Apps Script കോഡ് കാണാൻ ഇവിടെ ക്ലിക്ക് ചെയ്യുക 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("MadrasaBackup") || ss.insertSheet("MadrasaBackup");
  sheet.getRange("A1").setValue(new Date());
  sheet.getRange("B1").setValue(e.postData.contents);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("MadrasaBackup");
  var val = sheet ? sheet.getRange("B1").getValue() : "{}";
  return ContentService.createTextOutput(val || "{}").setMimeType(ContentService.MimeType.JSON);
}
