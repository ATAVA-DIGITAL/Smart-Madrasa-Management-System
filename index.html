Google Sheet Sync & JSON Backup

Download a full backup file to your device anytime, or restore from a saved JSON file.

View Google Apps Script (Code.gs) Setup Snippet 
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('MadrasaPortalData');
  if (!sheet) return ContentService.createTextOutput(JSON.stringify({ ok: true, data: null })).setMimeType(ContentService.MimeType.JSON);
  const raw = sheet.getRange('A1').getValue();
  return ContentService.createTextOutput(JSON.stringify({ ok: true, data: raw ? JSON.parse(raw) : null })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents || '{}');
  if (body.action === 'ping') {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, message: 'Connected to Google Sheet' })).setMimeType(ContentService.MimeType.JSON);
  }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName('MadrasaPortalData') || ss.insertSheet('MadrasaPortalData');
  sheet.getRange('A1').setValue(JSON.stringify(body.payload));
  sheet.getRange('B1').setValue(new Date().toISOString());
  return ContentService.createTextOutput(JSON.stringify({ ok: true, message: 'Synced to Google Sheet' })).setMimeType(ContentService.MimeType.JSON);
}
